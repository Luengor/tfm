\section{Introducción}

\begin{frame}{Contexto}
  \begin{columns}
    \column{0.65\textwidth}
      \begin{itemize}
        \item El grafiti no autorizado supone un \textbf{coste continuo} para las administraciones
        \item Seguimiento manual: \textbf{costoso e inconsistente}
        \item Agrupamiento automático por similitud visual $\Rightarrow$ trazabilidad de autores, priorización de retirada
      \end{itemize}
    \column{0.35\textwidth}
      \imgw{0.85}{graffiti_mosaic_vert.jpg}
  \end{columns}
\end{frame}

\begin{frame}{Problema y objetivo}
  \textbf{Problema:} dado un banco de fotografías de grafiti (mayoritariamente sin etiquetar), agruparlas por similitud visual de forma automática

  \vspace{1em}

  \textbf{Objetivo:} diseñar y evaluar un \textbf{sistema modular} que permita:
  \begin{itemize}
    \item Comparar combinaciones de extractores de características, técnicas de reducción y algoritmos de agrupamiento
    \item Caracterizar el \textbf{coste computacional} por etapa en función del tamaño del banco
    \item Evaluar la calidad del agrupamiento con métricas intrínsecas y extrínsecas
  \end{itemize}
\end{frame}

\end{frame}

\subsection{Extracción de características}
\begin{frame}{Extracción de características}
  \begin{columns}
    \column{0.5\textwidth}
      Modelos \textbf{preentrenados y congelados}: se toma su representación latente como vector de características.
      \vspace{0.6em}
      \begin{itemize}
        \item CNN: se elimina la capa de clasificación
        \item \textit{Transformers} (DINOv2, CLIP): ya devuelven un vector
        \item Vectores de entre $384$ y $4096$ dimensiones
      \end{itemize}
    \column{0.5\textwidth}
      \centering \small
      \begin{tabular}{llr}
        \toprule
        \textbf{Modelo} & \textbf{Tipo} & \textbf{Dim.} \\
        \midrule

        VGG16~\cite{simonyan_very_2015}         & CNN profunda      & $4096$ \\
        InceptionV3~\cite{szegedy_rethinking_2015}   & CNN multi-rama    & $2048$ \\
        MobileNetV3~\cite{howard_searching_2019}   & CNN ligera        & $1280$ \\
        DINOv2 ViT-S/14~\cite{oquab_dinov2_2024} & \textit{Transf.} & $384$ \\
        CLIP ViT-B/32~\cite{radford_learning_2021} & \textit{Transf.} & $512$ \\
        \bottomrule
      \end{tabular}
  \end{columns}
\end{frame}

\subsection{Reducción de dimensionalidad}
\begin{frame}{Reducción de dimensionalidad}
  Paso \textbf{opcional} entre almacenamiento y agrupamiento: reduce coste y mitiga la maldición de la dimensionalidad~\cite{aggarwal_surprising_2001}.
  \begin{columns}
    \column{0.54\textwidth}
      \begin{itemize}\small
        \item \textbf{PCA}~\cite{jolliffe_principal_2002} (lineal): direcciones de máxima varianza
        \item \textbf{Kernel PCA}~\cite{scholkopf_nonlinear_1998} (no lineal): PCA en un espacio de características implícito
        \item \textbf{Isomap}~\cite{tenenbaum_global_2000} (no lineal): preserva distancias geodésicas sobre el grafo de vecinos
        \item \textbf{UMAP}~\cite{mcinnes_umap_2020} (no lineal): preserva la estructura local de vecindad
      \end{itemize}
      \vspace{0.3em}
      {\footnotesize Las métricas de calidad se calculan sobre los \textit{embeddings} \textbf{originales} (sin reducir).}
    \column{0.46\textwidth}
      \imgw{1.0}{curse_dimensionality.pdf}\\
      \centering \scriptsize Conforme aumenta el número de dimensiones la
      distancia entre el vecino más cercano y más lejano se vuelven
      equivalentes.
  \end{columns}
\end{frame}

\subsection{Búsqueda por similitud}
\begin{frame}{Búsqueda por similitud: recuperación basada en contenido}
  El mismo espacio de \textit{embeddings} habilita la \textbf{recuperación de
  imágenes basada en contenido} (\textit{content-based image
  retrieval}): dada una firma de consulta, devolver las visualmente más
  parecidas~\cite{van_rijsbergen_c_j_information_1979,fiel_writer_2015}.
  \begin{columns}[c]
    \column{0.5\textwidth}
      \centering
      \shorthandoff{<>}%
      \begin{tikzpicture}[
        >=Stealth, node distance=6mm and 12mm,
        block/.style={draw, rounded corners, minimum height=7mm, minimum width=18mm,
                      inner sep=3pt, align=center, font=\scriptsize},
        idx/.style={block, fill=green!12},
        res/.style={block, fill=blue!8, minimum width=12mm},
        lbl/.style={font=\scriptsize, midway, fill=white, inner sep=1pt},
      ]
        \node[block] (q) {consulta};
        \node[idx, right=of q] (i) {índice de\\\textit{embeddings}};
        \node[res, right=14mm of i] (r1) {vecino 1};
        \node[res, below=3mm of r1] (r2) {vecino 2};
        \node[res, below=3mm of r2] (r3) {\dots};
        \draw[->] (q)--(i) node[lbl,above]{\textit{embed.}};
        \draw[->] (i.east)--(r1.west) node[lbl,above]{top-$k$};
        \draw[->] (i.east)--(r2.west);
        \draw[->] (i.east)--(r3.west);
      \end{tikzpicture}%
      \shorthandon{<>}
    \column{0.5\textwidth}
      \begin{itemize}\small
        \item Dos \textit{backends}: barrido \textbf{exacto} (SQLite, fuerza bruta) vs.\ índice
          aproximado \textbf{HNSW}~\cite{malkov_efficient_2018} sobre pgvector
        \item Compromiso \textbf{velocidad / exactitud}: el ANN sacrifica
          recuperar el vecino \emph{exacto} por escalar a bancos
          grandes~\cite{johnson_billion-scale_2017}
        \item Calidad con métricas clásicas de RI: \textbf{P@$k$, R@$k$,
          MAP@$k$, MRR}~\cite{van_rijsbergen_c_j_information_1979}
      \end{itemize}
      \vspace{0.3em}
      {\footnotesize Es a la vez \textbf{herramienta de consulta} y \textbf{sonda de
      calidad} del espacio de \textit{embeddings} (sección de resultados).}
  \end{columns}
\end{frame}

        \item \textbf{Intrínsecas} (sin etiquetas): silueta~\cite{rousseeuw_silhouettes_1987}, Calinski--Harabasz~\cite{calinski_dendrite_1974}, Davies--Bouldin~\cite{davies_cluster_1979}, ruido, balance de tamaños
        \item \textbf{Extrínsecas} (con etiquetas): ARI~\cite{hubert_comparing_1985}, NMI~\cite{strehl_cluster_2002}, F1 pareada
        \item \textbf{Coste}: tiempo de pared y memoria pico por etapa
      \end{itemize}
      \vspace{0.3em}
      \begin{block}{\scriptsize Configuración base resultante}
        {\scriptsize DINOv2 cabeza-estilo $+$ UMAP-10 $+$ HDBSCAN (\texttt{mcs}=5) $+$ segmentador identidad $+$ SQLite}
      \end{block}
      {\scriptsize Las métricas intrínsecas se leen \textbf{en conjunto}: aisladas premian particiones degeneradas.}
  \end{columns}
\end{frame}

\begin{frame}{Resultados: calidad intrínseca y elección de componentes}
  \begin{columns}[c]
    \column{0.52\textwidth}
      \centering
      \includegraphics[width=\linewidth]{figures/e3_pareto_quality_cost.pdf}\\
      {\scriptsize Calidad frente a coste por familia de agrupamiento ($N{=}6416$).}
    \column{0.48\textwidth}
      \footnotesize
      \begin{itemize}
        \item \textbf{HDBSCAN} domina el frente de Pareto: silueta competitiva, ruido moderado, tamaños balanceados y coste despreciable
        \item OPTICS lo iguala en calidad pero paga $\sim$25$\times$ tiempo y $\sim$2500$\times$ memoria
        \item \textbf{UMAP-10} es la única reducción no degenerada; la identidad descarta $\sim$68\,\% de los puntos como ruido
        \item Silueta y CH \textbf{aisladas engañan}: premian $k{=}2$ trivial $\Rightarrow$ criterio conjunto
      \end{itemize}
  \end{columns}
\end{frame}

\section{Conclusiones}

\begin{frame}{Conclusiones}
  \begin{columns}[t]
    \column{0.5\textwidth}
      \textbf{Hallazgos principales}
      \begin{itemize}\small
        \item Sistema \textbf{modular} que compara extractores, reducciones y agrupadores bajo una misma interfaz
        \item \textbf{Pipeline base}: cabeza de estilo DINOv2 $+$ UMAP-10 $+$ HDBSCAN; valida la cabeza por su calidad \textbf{supervisada}, no la intrínseca
        \item El \textbf{coste} lo dominan ingesta y, sobre todo, la búsqueda $O(n^2)$; \textbf{HNSW} la elimina sin perder \textit{recall}
        \item La reducción de dimensionalidad es \textbf{indispensable} para calidad y coste
      \end{itemize}
    \column{0.5\textwidth}
      \textbf{Limitaciones y trabajo futuro}
      \begin{itemize}\small
        \item Rango de $N$ estrecho ($\le 6416$): difícil separar $O(n\log n)$ de $O(n^2)$
        \item Datos de \textbf{autoría} insuficientes $\Rightarrow$ ampliar y anotar el banco
        \item Preentrenamiento no supervisado antes de las cabezas supervisadas
        \item Pasar de \textit{batch} a \textbf{ingesta incremental} (índice y espacio reducido)
      \end{itemize}
  \end{columns}
  \vfill
  \centering
  \footnotesize El trabajo \textbf{caracteriza} desempeño y coste de la similitud visual en bancos de grafiti, con una recomendación concreta y reproducible.
