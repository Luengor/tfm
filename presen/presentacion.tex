\section{Introducción}

\begin{frame}{Contexto}
  \begin{columns}
    \column{0.65\textwidth}
      \begin{itemize}
        \item El grafiti no autorizado es un fenómeno urbano recurrente que
            requiere atención continua por parte de las administraciones
        \item El seguimiento manual es \textbf{costoso e inconsistente}
        \item Un agrupamiento automático por similitud visual facilitaría
            la trazabilidad de autores o la priorización de retirada basada
            en el contenido visual
      \end{itemize}
    \column{0.35\textwidth}
      \imgw{0.85}{graffiti_mosaic_vert.jpg}
  \end{columns}
\end{frame}

\begin{frame}{Problema y objetivo}
  \textbf{Problema:} dado un banco de fotografías de grafiti sin etiquetar,
  agruparlas por similitud visual de forma automática

  \vspace{1em}

  \textbf{Objetivo:} diseñar y evaluar un \textbf{sistema modular} que permita:
  \begin{itemize}
    \item Comparar combinaciones de extractores de características, técnicas de reducción y algoritmos de agrupamiento
    \item Caracterizar el \textbf{coste computacional} por etapa en función del tamaño del banco
    \item Evaluar la calidad del agrupamiento con métricas intrínsecas y extrínsecas
  \end{itemize}
\end{frame}

\begin{frame}{Trabajos relacionados}
  \begin{table}
    \centering
    \large
    \begin{tabular}{lll}
      \toprule
      \textbf{Trabajo} & \textbf{Tarea} & \textbf{Limitación} \\
      \midrule

      García García et al.~\cite{garcia_garcia_graffiti_2025} & Agrupamiento por color & Sin descriptores profundos \\
      \midrule

      \bottomrule
    \end{tabular}
  \end{table}

  \vspace{0.5em}
  \centering
  \small Ningún trabajo previo combina \textbf{descriptores profundos}, agrupamiento clásico\\
  y caracterización del \textbf{coste computacional} sobre imágenes de grafiti.
\end{frame}

\section{Conjunto de datos}

\begin{frame}{Conjunto de datos}
  \begin{columns}
    \column{0.55\textwidth}
      \begin{itemize}
        \item \textbf{Banco principal:} $6681$ fotografías de grafiti en
            \textbf{Salamanca}
        \item Capturadas con un teléfono móvil ($\sim$10.6~MP), sin
            postprocesado. Captura masiva en otoño de 2022

\subsection{Extracción de características}
\begin{frame}{Extracción de características}
  \begin{columns}
    \column{0.5\textwidth}
      Modelos \textbf{preentrenados y congelados}: se toma su representación latente como vector de características.
      \vspace{0.6em}
      \begin{itemize}
        \item CNN: se elimina la capa de clasificación
        \item \textit{Transformers} (DINOv2, CLIP): ya devuelven un vector
        \item Vectores de entre $384$ y $4096$ dimensiones
      \end{itemize}
    \column{0.5\textwidth}
      \centering \small
      \begin{tabular}{llr}
        \toprule
        \textbf{Modelo} & \textbf{Tipo} & \textbf{Dim.} \\
        \midrule
        ResNet50      & CNN residual      & $2048$ \\
        VGG16         & CNN profunda      & $4096$ \\
        InceptionV3   & CNN multi-rama    & $2048$ \\
        MobileNetV3   & CNN ligera        & $1280$ \\
        DINOv2 ViT-S/14 & \textit{Transf.} & $384$ \\
        CLIP ViT-B/32 & \textit{Transf.} & $512$ \\
        \bottomrule
      \end{tabular}
  \end{columns}
\end{frame}

\begin{frame}{Extracción de características: ajuste al dominio}
  \begin{columns}
    \column{0.65\textwidth}

      \begin{itemize}
        \item \textit{Backbone} \textbf{congelado}, solo se entrena una cabeza
            ligera (Linear→ReLU→Linear→$L_2$)
        \item Bases: \textbf{MobileNetV3} y \textbf{DINOv2}
        \item Tareas: \textbf{autoría} (pérdida de tripletes) y \textbf{estilo}
            (\textit{SupCon})
        \item En total 4 cabezas
      \end{itemize}

\subsection{Reducción de dimensionalidad}
\begin{frame}{Reducción de dimensionalidad}
  Paso \textbf{opcional} entre almacenamiento y agrupamiento: reduce coste y mitiga la maldición de la dimensionalidad~\cite{aggarwal_surprising_2001}.
  \begin{columns}
    \column{0.54\textwidth}
      \begin{itemize}\small
        \item \textbf{PCA} (lineal): direcciones de máxima varianza
        \item \textbf{Kernel PCA} (no lineal): PCA en un espacio de
            características implícito
        \item \textbf{Isomap} (no lineal): preserva distancias geodésicas sobre
            el grafo de vecinos
        \item \textbf{UMAP} (no lineal): preserva la estructura local de
            vecindad
      \end{itemize}
      \vspace{0.3em}
      {\footnotesize Las métricas de calidad se calculan sobre los \textit{embeddings} \textbf{originales} (sin reducir).}
    \column{0.46\textwidth}
      \imgw{1.0}{curse_dimensionality.pdf}\\
      \centering \scriptsize Conforme aumenta el número de dimensiones las
      distancias entre el vecino más cercano y más lejano se vuelven
      equivalentes.
  \end{columns}
\end{frame}

\subsection{Agrupamiento}
\begin{frame}{Agrupamiento}
  \begin{columns}

      \vspace{0.5em}

        \item \textbf{Particional:} K-medias
        \item \textbf{Densidad:} DBSCAN, OPTICS, HDBSCAN
        \item \textbf{Jerárquico:} Aglomerativo
        \item \textbf{Grafo:} Espectral
        \item \textbf{Probabilístico:} GMM
        \item \textbf{Mensajes:} Affinity Propagation
      \end{itemize}

\begin{frame}{Resultados: reducción de dimensionalidad (E2)}
  \footnotesize
  \begin{itemize}
    \item Sin reducción: presenta la mejor silueta del experimento, pero a
        costa de un $68\%$ de ruido. El coste de memoria y tiempo se trasladan de la etapa
        de reducción al agrupamiento
    \item PCA solo produce agrupamientos degenerados: dos clústeres y silueta nula
        en PCA-10 y un $69.8\%$ de ruido en PCA-50
    \item \textbf{UMAP}: Mejor equilibrio entre silueta y ruido. UMAP-50 da
        resultados prácticamente idénticos a UMAP-10, pero prácticamente
        doblando el coste de memoria y tiempo
  \end{itemize}

        \item \textbf{HDBSCAN}: silueta competitiva, ruido moderado, tamaños balanceados, coste despreciable ($0.35$~s)
        \item OPTICS iguala en calidad, ${\sim}25\times$ más tiempo, ${\sim}2\,500\times$ más memoria
        \item K-medias: más rápido, pero silueta 0.094 puntos inferior. Mejor opción sin ruido
        \item Métricas \textbf{aisladas} engañan: silueta y CH premian $k{=}2$ trivial $\Rightarrow$ criterio conjunto
      \end{itemize}
      \vspace{0.4em}
      \begin{block}{\scriptsize Pipeline base}
        {\scriptsize DINOv2 estilo $+$ UMAP-10 $+$ HDBSCAN (\texttt{mcs}=5) $+$ identidad}
      \end{block}
  \end{columns}
\end{frame}

\section{Conclusiones}

\begin{frame}{Conclusiones}
  \begin{columns}[t]
    \column{0.5\textwidth}
      \textbf{Hallazgos principales}
      \begin{itemize}\small
        \item Sistema \textbf{modular} que compara extractores, reducciones y
            agrupadores bajo una misma interfaz
        \item \textbf{Pipeline base}: cabeza de estilo DINOv2 $+$ UMAP-10 $+$
            HDBSCAN; valida la cabeza por su calidad \textbf{supervisada}, no
            la intrínseca
        \item El \textbf{coste} lo dominan ingesta y, sobre todo, la búsqueda
            $O(n^2)$; \textbf{HNSW} la elimina sin perder \textit{recall}
        \item La reducción de dimensionalidad es \textbf{indispensable} para
            calidad y coste
      \end{itemize}
    \column{0.5\textwidth}
      \textbf{Limitaciones y trabajo futuro}
      \begin{itemize}\small
        \item Rango de $N$ estrecho ($\le 6416$): difícil separar $O(n\log n)$ de $O(n^2)$
        \item Datos de \textbf{autoría} insuficientes $\Rightarrow$ ampliar y anotar el banco
        \item Preentrenamiento no supervisado antes de las cabezas supervisadas
        \item Pasar de \textit{batch} a \textbf{ingesta incremental} (índice y espacio reducido)
      \end{itemize}
  \end{columns}

\end{frame}
