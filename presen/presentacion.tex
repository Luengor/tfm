\section{Introducción}

\begin{frame}{Contexto}
  \begin{columns}
    \column{0.65\textwidth}
      \begin{itemize}
        \item El grafiti no autorizado supone un \textbf{coste continuo} para las administraciones
        \item Seguimiento manual: \textbf{costoso e inconsistente}
        \item Agrupamiento automático por similitud visual $\Rightarrow$ trazabilidad de autores, priorización de retirada
      \end{itemize}
    \column{0.35\textwidth}
      \imgw{0.85}{graffiti_mosaic_vert.jpg}
  \end{columns}
\end{frame}

\begin{frame}{Problema y objetivo}
  \textbf{Problema:} dado un banco de fotografías de grafiti (mayoritariamente sin etiquetar), agruparlas por similitud visual de forma automática

  \vspace{1em}

  \textbf{Objetivo:} diseñar y evaluar un \textbf{sistema modular} que permita:
  \begin{itemize}
    \item Comparar combinaciones de extractores de características, técnicas de reducción y algoritmos de agrupamiento
    \item Caracterizar el \textbf{coste computacional} por etapa en función del tamaño del banco
    \item Evaluar la calidad del agrupamiento con métricas intrínsecas y extrínsecas
  \end{itemize}
\end{frame}

\begin{frame}{Trabajos relacionados}
  \begin{table}
    \centering
    \large
    \begin{tabular}{lll}
      \toprule
      \textbf{Trabajo} & \textbf{Tarea} & \textbf{Limitación} \\
      \midrule
      Fogaça et al.~\cite{fogaca_deep_2023}             & Detección y clasificación  & Supervisado, sin agrupamiento \\
      Tokuda et al.~\cite{tokuda_quantifying_2019}      & Cuantificación geográfica  & Sin similitud visual \\
      García García et al.~\cite{garcia_garcia_graffiti_2025} & Agrupamiento por color & Sin descriptores profundos \\
      \midrule
      \textbf{Este trabajo} & \textbf{Agrupamiento + coste} & --- \\
      \bottomrule
    \end{tabular}
  \end{table}

  \vspace{0.5em}
  \centering
  \small Ningún trabajo previo combina \textbf{descriptores profundos}, agrupamiento clásico\\
  y caracterización del \textbf{coste computacional} sobre imágenes de grafiti.
\end{frame}

\section{Conjunto de datos}

\begin{frame}{Conjunto de datos}
  \begin{columns}
    \column{0.55\textwidth}
      \begin{itemize}
        \item \textbf{Banco principal:} $6681$ fotografías de grafiti en
            \textbf{Salamanca}
        \item Capturadas con dispositivo móvil ($\sim$10.6~MP), sin
            postprocesado. Captura masiva en otoño de 2022
        \item Alta variabilidad entre fotografías: iluminación, perspectiva,
            oclusión, escena, etc. Muchas de las fotografías contienen
            \textbf{múltiples grafitis}
        \item \textbf{Banco auxiliar:} $1106$ imágenes de \textbf{Cuenca}
            (StopGrafiti) $\Rightarrow$ solo ajuste fino, evita solapamiento
            entrenamiento/evaluación
      \end{itemize}
    \column{0.45\textwidth}
      \imgw{1.0}{map_with_photos.pdf}
      \centering \scriptsize Ubicación de las imágenes con GPS
  \end{columns}
\end{frame}

\subsection{Segmentación}
\begin{frame}{Segmentación}
  \begin{columns}
    \column{0.42\textwidth}
      \begin{itemize}
        \item Etapa \textbf{opcional} para recortar grafitis de la escena
        \item Detector \textbf{YOLO ajustado} sobre grafiti ($447$ imágenes, $\sim$650 cajas; partida de pesos COCO)
        \item \textbf{Fusión} de cajas por solapamiento y proximidad; salvaguarda anti-mega-cajas
        \item \textbf{Margen} de contexto configurable antes de recortar
      \end{itemize}
    \column{0.58\textwidth}
      \imgw{1.0}{segmentation_stages.pdf}\\
      \centering \scriptsize Original → detecciones → fusión → margen → recortes
  \end{columns}
\end{frame}

\subsection{Extracción de características}
\begin{frame}{Extracción de características}
  \begin{columns}
    \column{0.5\textwidth}
      Modelos \textbf{preentrenados y congelados}: se toma su representación latente como vector de características.
      \vspace{0.6em}
      \begin{itemize}
        \item CNN: se elimina la capa de clasificación
        \item \textit{Transformers} (DINOv2, CLIP): ya devuelven un vector
        \item Vectores de entre $384$ y $4096$ dimensiones
      \end{itemize}
    \column{0.5\textwidth}
      \centering \small
      \begin{tabular}{llr}
        \toprule
        \textbf{Modelo} & \textbf{Tipo} & \textbf{Dim.} \\
        \midrule
        ResNet50      & CNN residual      & $2048$ \\
        VGG16         & CNN profunda      & $4096$ \\
        InceptionV3   & CNN multi-rama    & $2048$ \\
        MobileNetV3   & CNN ligera        & $1280$ \\
        DINOv2 ViT-S/14 & \textit{Transformer} & $384$ \\
        CLIP ViT-B/32 & \textit{Transformer} & $512$ \\
        \bottomrule
      \end{tabular}
  \end{columns}
\end{frame}

\begin{frame}{Extracción de características: ajuste al dominio}
  \begin{columns}
    \column{0.65\textwidth}
      \textbf{Cabezas de proyección} entrenadas sobre grafiti utilizando \textit{transfer learning}:
      \begin{itemize}
        \item \textit{Backbone} \textbf{congelado}, solo se entrena una cabeza ligera (Linear→ReLU→Linear→$L_2$)
        \item Bases: \textbf{MobileNetV3} y \textbf{DINOv2}
        \item Tareas: \textbf{autoría} (pérdida de tripletes) y \textbf{estilo} (\textit{SupCon})
        \item En total 4 cabezas
      \end{itemize}
    \column{0.35\textwidth}
      \centering
      \shorthandoff{<>}%
      \hspace*{1.5em}%
      \begin{tikzpicture}[
        >=Stealth, node distance=4.5mm,
        block/.style={draw, rounded corners, minimum height=7mm, minimum width=30mm,
                      inner sep=3pt, align=center, font=\small},
        frozen/.style={block, fill=gray!20, dashed},
        trained/.style={block, fill=blue!12},
        vec/.style={font=\footnotesize, align=center},
      ]
        \node[vec] (img) {recorte};
        \node[frozen, below=of img] (bb) {\textit{Backbone} congelado};
        \node[trained, below=of bb] (l1) {Lineal};
        \node[trained, below=of l1] (relu) {ReLU};
        \node[trained, below=of relu] (l2) {Lineal};
        \node[block, fill=green!12, below=of l2] (norm) {$L_2$};
        \node[vec, below=of norm] (e) {\textit{embedding} ($128$)};
        \draw[->] (img)--(bb); \draw[->] (bb)--(l1); \draw[->] (l1)--(relu);
        \draw[->] (relu)--(l2); \draw[->] (l2)--(norm); \draw[->] (norm)--(e);
        \node[draw, dotted, fit=(l1)(relu)(l2), inner sep=1.5mm,
              label={[align=center,font=\scriptsize]right:{cabeza\\entrenada}}] {};
      \end{tikzpicture}%

  \end{columns}
\end{frame}

\subsection{Reducción de dimensionalidad}
\begin{frame}{Reducción de dimensionalidad}
  Paso \textbf{opcional} entre almacenamiento y agrupamiento: reduce coste y mitiga la maldición de la dimensionalidad~\cite{aggarwal_surprising_2001}.
  \begin{columns}
    \column{0.54\textwidth}
      \begin{itemize}\small
        \item \textbf{PCA} (lineal): direcciones de máxima varianza
        \item \textbf{Kernel PCA} (no lineal): PCA en un espacio de características implícito
        \item \textbf{Isomap} (no lineal): preserva distancias geodésicas sobre el grafo de vecinos
        \item \textbf{UMAP} (no lineal): preserva la estructura local de vecindad
      \end{itemize}
      \vspace{0.3em}
      {\footnotesize Las métricas de calidad se calculan sobre los \textit{embeddings} \textbf{originales} (sin reducir).}
    \column{0.46\textwidth}
      \imgw{1.0}{curse_dimensionality.pdf}\\
      \centering \scriptsize Conforme aumenta el número de dimensiones la
      distancia entre el vecino más cercano y más lejano se vuelven
      equivalentes.
  \end{columns}
\end{frame}
